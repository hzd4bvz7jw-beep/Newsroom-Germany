const Parser = require('rss-parser'); const fs = require('fs'); // Parser mit vollwertigem User-Agent (verhindert Blockaden durch Google/Simracing-Feeds) const parser = new Parser({ headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36', 'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8' } }); // Definition aller Feed-Quellen const FEEDS = { de: [ 'https://www.gamestar.de/news/rss/news.rss', 'https://www.computerbase.de/rss/news.xml', 'https://www.pcgameshardware.de/feed.cfm' ], sim: [ 'https://news.google.com/rss/search?q=Simracing+OR+iRacing+OR+%22Assetto+Corsa%22&hl=de&gl=DE&ceid=DE:de', 'https://www.speedweek.com/rss/news/1', 'https://www.motorsport-total.com/rss/news.xml' ], int: [ 'https://feeds.feedburner.com/ign/all', 'https://www.eurogamer.net/feed/news' ] }; // Hilfsfunktion: Entfernt HTML-Tags aus Beschreibungen function stripHtml(html) { if (!html) return ''; let text = html.replace(/<[^>]*>?/gm, ''); return text.length > 150 ? text.substring(0, 150) + '...' : text; } // Lädt einen einzelnen RSS-Feed async function loadFeed(url) { try { const feed = await parser.parseURL(url); const source = url.includes('google.com') ? 'Simracing DE' : (feed.title || 'News').replace(/ Feed| RSS/g, ''); return feed.items.map(item => ({ title: item.title, link: item.link, pubDate: item.pubDate ? new Date(item.pubDate) : new Date(), snippet: stripHtml(item.contentSnippet || item.content || item.summary), source: source })); } catch (err) { console.error(`Fehler bei ${url}:`, err.message); return []; } } // Hauptfunktion: Feeds verarbeiten und index.html generieren async function build() { const categories = {}; for (const [key, urls] of Object.entries(FEEDS)) { let items = []; for (const url of urls) { const fetched = await loadFeed(url); items = items.concat(fetched); } // Duplikate filtern & nach Datum sortieren const unique = Array.from(new Map(items.map(item => [item.title.toLowerCase().trim(), item])).values()); unique.sort((a, b) => b.pubDate - a.pubDate); categories[key] = unique.slice(0, 12); } function renderCards(items) { if (!items || items.length === 0) return '
Keine News verfügbar.
'; return items.map(item => `

${item.title}

📅 ${item.pubDate.toLocaleDateString('de-DE', {day: '2-digit', month:'2-digit'})} ${item.pubDate.toLocaleTimeString('de-DE', {hour: '2-digit', minute:'2-digit'})} Uhr 📌 ${item.source}
${item.snippet}
`).join(''); } // Generiert das vollständige HTML-Dokument const htmlContent = ` Gaming & Simracing News Dashboard

🎮 Gaming & Simracing Live Dashboard

Zuletzt aktualisiert: ${new Date().toLocaleString('de-DE', { timeZone: 'Europe/Berlin' })} Uhr
🇩🇪 Gaming (DACH)
${renderCards(categories.de)}
🏎️ Simracing (DE)
${renderCards(categories.sim)}
🌐 Gaming (International)
${renderCards(categories.int)}
`; fs.writeFileSync('index.html', htmlContent); console.log('index.html wurde erfolgreich neu generiert!'); } build();
